```java
@SpringBootTest
@ActiveProfiles("test")
class AIServiceIntegrationTest {

    @Autowired
    private RestTemplate restTemplate;

    @Autowired
    private AIServiceClient aiServiceClient;

    private MockRestServiceServer mockServer;

    @BeforeEach
    void setUp() {
        // 模拟AI-Service响应
        mockServer = MockRestServiceServer.createServer(restTemplate);
    }

    @Test
    @DisplayName("场景1: AI生成测试用例 - 成功")
    void generateTestCases_Success() throws Exception {
        // Given: 准备Mock响应
        AITestCaseGenerationResponse mockResponse = new AITestCaseGenerationResponse();
        mockResponse.setSuccess(true);
        mockResponse.setGeneratedCases(Arrays.asList(/* ... */));

        mockServer.expect(requestTo(aiServiceUrl + "/api/v1/ai/testcase/generate"))
                .andExpect(method(HttpMethod.POST))
                .andRespond(withSuccess(
                    objectMapper.writeValueAsString(mockResponse),
                    MediaType.APPLICATION_JSON
                ));

        // When: 调用AIServiceClient
        AITestCaseGenerationResponse response = aiServiceClient.generateTestCases(request);

        // Then: 验证响应
        assertNotNull(response);
        assertTrue(response.isSuccess());
        assertEquals(2, response.getTotalCount());
    }

    @Test
    @DisplayName("场景2: AI服务不可用 - 触发Fallback")
    void generateTestCases_ServiceUnavailable_FallbackTriggered() {
        // Given: Mock 500错误
        mockServer.expect(requestTo(aiServiceUrl + "/api/v1/ai/testcase/generate"))
                .andExpect(method(HttpMethod.POST))
                .andRespond(withServerError());

        // When: 调用AIServiceClient（会触发Fallback）
        AITestCaseGenerationResponse response = aiServiceClient.generateTestCases(request);

        // Then: 验证Fallback响应
        assertNotNull(response);
        assertFalse(response.isSuccess());
        assertTrue(response.getMessage().contains("AI服务暂时不可用"));
    }
}
```

import { test, expect } from '@playwright/test';

test.describe('策略推荐功能', () => {
  test('完整流程: 输入信息 → 获取推荐 → 查看详情', async ({ page }) => {

## 故障排查

### 常见问题

#### 1. 测试数据库连接失败

**问题**: `java.sql.SQLException: Access denied for user 'test_user'@'localhost'`

**解决方案**:
```sql
-- 创建测试用户并授权
CREATE USER 'test_user'@'localhost' IDENTIFIED BY 'test_password';
GRANT ALL PRIVILEGES ON synapsetest_test.* TO 'test_user'@'localhost';
FLUSH PRIVILEGES;
```

#### 2. MockRestServiceServer验证失败

**问题**: `AssertionError: Further request(s) expected but not found`

**原因**: 测试期望有请求但未发生

**解决方案**:
```java
@AfterEach
void tearDown() {
    // 不要自动验证，手动控制
    // mockServer.verify();
}

@Test
void test() {
    // 测试逻辑

    // 显式验证
    mockServer.verify();
}
```

#### 3. AI-Service测试超时

**问题**: 测试运行时间过长

**解决方案**:
```python
# 使用Mock LLM提供商
import pytest
from unittest.mock import Mock

@pytest.fixture
def mock_llm():
    mock = Mock()
    mock.generate.return_value = "生成的测试用例内容"
    return mock

def test_with_mock_llm(mock_llm):
    # 使用Mock LLM，避免真实API调用
    service = TestCaseGenerationService(llm_provider=mock_llm)
    result = service.generate(...)
```

#### 4. Frontend测试找不到元素

**问题**: `Unable to find an element with the text: ...`

**解决方案**:
```typescript
// 使用waitFor等待异步操作
import { waitFor } from '@testing-library/react';

await waitFor(() => {
  expect(screen.getByText('期望的文本')).toBeInTheDocument();
}, { timeout: 3000 });

// 或使用findBy* (内置waitFor)
const element = await screen.findByText('期望的文本');
```

## 最佳实践

### 1. 测试命名规范

```java
// Java: Given_When_Then模式
@Test
void createTestCase_WithValidData_ShouldReturnCreatedCase() { }

// Python: test_<功能>_<场景>_<期望结果>
def test_recommend_strategy_prod_environment_returns_full_scope(): pass

// TypeScript: should + 期望行为
it('should display error message when API fails', () => { });
```

### 2. 测试数据构建

```java
// 使用Builder模式
TestCase testCase = TestCase.builder()
    .caseName("测试用例")
    .priority(8)
    .type("FUNCTIONAL")
    .build();

// 使用工厂方法
TestCase testCase = TestCaseFactory.createDefault();
```

### 3. 断言清晰性

```java
// ❌ 不好的断言
assertTrue(result.getCases().size() > 0);

// ✅ 好的断言
assertThat(result.getCases())
    .isNotEmpty()
    .hasSizeGreaterThan(0)
    .allMatch(c -> c.getConfidence() > 0.8);
```

### 4. 测试隔离

```java
// 每个测试都应该独立运行
@BeforeEach
void setUp() {
    // 重置状态
    testDataRepository.deleteAll();
}

@AfterEach
void tearDown() {
    // 清理资源
    mockServer.reset();
}
```
