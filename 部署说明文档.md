## 2. 本地开发环境部署

### 2.1 依赖服务启动

#### 2.1.1 MySQL 8.0

**方式1: 使用Docker启动**

```bash
# 启动MySQL容器
docker run -d \
  --name mysql-testdb \
  -p 3306:3306 \
  -e MYSQL_ROOT_PASSWORD=root123 \
  -e MYSQL_DATABASE=synapse_test \
  -v mysql-data:/var/lib/mysql \
  mysql:8.0

# 验证MySQL运行状态
docker ps | grep mysql-testdb
```

**方式2: 本地安装**

```bash
# macOS
brew install mysql@8.0
brew services start mysql@8.0

# Ubuntu/Debian
sudo apt update
sudo apt install mysql-server
sudo systemctl start mysql

# 创建数据库
mysql -u root -p
CREATE DATABASE synapse_test DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
CREATE USER 'synapse'@'localhost' IDENTIFIED BY 'synapse123';
GRANT ALL PRIVILEGES ON synapse_test.* TO 'synapse'@'localhost';
FLUSH PRIVILEGES;
EXIT;
```

**初始化数据库表结构**

```bash
# 执行SQL初始化脚本
mysql -u synapse -p synapse_test < backend/sql/schema.sql
mysql -u synapse -p synapse_test < backend/sql/init_data.sql
```

**方式1: 使用Docker启动**

```bash
# 启动Redis容器
docker run -d \
  --name redis-cache \
  -p 6379:6379 \
```

## 3. Docker部署

### 3.1 构建Docker镜像

#### 3.1.1 构建前端镜像

```bash
# 进入前端目录
cd frontend

# 构建镜像
docker build -t synapse-frontend:latest .

# 验证镜像
docker images | grep synapse-frontend
```

**Dockerfile 说明**:

```dockerfile
# frontend/Dockerfile

# 构建阶段
FROM node:18-alpine AS builder
WORKDIR /app
COPY package*.json ./
RUN npm ci
COPY . .
RUN npm run build

# 生产阶段
FROM nginx:alpine
COPY --from=builder /app/dist /usr/share/nginx/html
COPY nginx.conf /etc/nginx/conf.d/default.conf
EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]
```

**nginx.conf 配置**:

```nginx
# frontend/nginx.conf

server {
    listen 80;
    server_name localhost;
    root /usr/share/nginx/html;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }

    location /api/ {
        proxy_pass http://backend:8080/api/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    location /ai-api/ {
        proxy_pass http://ai-service:8000/api/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    location /ws {
        proxy_pass http://backend:8080/ws;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }
}
```

### 3.2 Docker Compose部署

#### 3.2.1 完整部署 (MySQL + Qdrant)

**推荐配置**: 使用 `docker-compose.mysql-qdrant.yml`

```bash
# 启动所有服务
docker-compose -f docker-compose.mysql-qdrant.yml up -d

# 查看服务状态
docker-compose -f docker-compose.mysql-qdrant.yml ps

# 查看日志
docker-compose -f docker-compose.mysql-qdrant.yml logs -f

# 停止所有服务
docker-compose -f docker-compose.mysql-qdrant.yml down

# 停止并删除数据卷
docker-compose -f docker-compose.mysql-qdrant.yml down -v
```

**docker-compose.mysql-qdrant.yml 配置**:

```yaml
version: '3.8'

services:
  # MySQL数据库
  mysql:
    image: mysql:8.0
    container_name: synapse-mysql
    environment:
      MYSQL_ROOT_PASSWORD: root123
      MYSQL_DATABASE: synapse_test
      MYSQL_USER: synapse
      MYSQL_PASSWORD: synapse123
    ports:
      - "3306:3306"
    volumes:
      - mysql-data:/var/lib/mysql
      - ./backend/sql:/docker-entrypoint-initdb.d
    command: --default-authentication-plugin=mysql_native_password
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - synapse-network

  # Qdrant向量数据库
  qdrant:
    image: qdrant/qdrant:v1.7.4
    container_name: synapse-qdrant
    ports:
      - "6333:6333"
      - "6334:6334"
    volumes:
      - qdrant-storage:/qdrant/storage
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:6333/"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - synapse-network

  # Redis缓存
  redis:
    image: redis:7-alpine
    container_name: synapse-redis
    ports:
      - "6379:6379"
    volumes:
      - redis-data:/data
    command: redis-server --appendonly yes
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - synapse-network

  # AI服务
  ai-service:
    build:
      context: ./ai-service
      dockerfile: Dockerfile
    image: synapse-ai-service:latest
    container_name: synapse-ai-service
    environment:
      MYSQL_HOST: mysql
      MYSQL_PORT: 3306
      MYSQL_USER: synapse
      MYSQL_PASSWORD: synapse123
      MYSQL_DATABASE: synapse_test
      REDIS_HOST: redis
      REDIS_PORT: 6379
      VECTOR_DB_TYPE: qdrant
      QDRANT_HOST: qdrant
      QDRANT_PORT: 6333
      LLM_PROVIDER: qwen
      LLM_DEVICE: cpu
      BACKEND_URL: http://backend:8080
    ports:
      - "8000:8000"
    volumes:
      - ./ai-service/models:/app/models
      - ./ai-service/logs:/app/logs
    depends_on:
      mysql:
        condition: service_healthy
      qdrant:
        condition: service_healthy
      redis:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 60s
    networks:
      - synapse-network

  # 后端服务
  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    image: synapse-backend:latest
    container_name: synapse-backend
    environment:
      SPRING_PROFILES_ACTIVE: prod
      SPRING_DATASOURCE_URL: jdbc:mysql://mysql:3306/synapse_test?useUnicode=true&characterEncoding=utf-8&useSSL=false&serverTimezone=Asia/Shanghai
      SPRING_DATASOURCE_USERNAME: synapse
      SPRING_DATASOURCE_PASSWORD: synapse123
      SPRING_REDIS_HOST: redis
      SPRING_REDIS_PORT: 6379
      AI_SERVICE_URL: http://ai-service:8000
    ports:
      - "8080:8080"
    depends_on:
      mysql:
        condition: service_healthy
      redis:
        condition: service_healthy
      ai-service:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8080/actuator/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 40s
    networks:
      - synapse-network

  # 前端服务
  frontend:
    build:
      context: ./frontend
      dockerfile: Dockerfile
    image: synapse-frontend:latest
    container_name: synapse-frontend
    ports:
      - "80:80"
    depends_on:

        condition: service_healthy
```

## 5. 常见问题排查

### 5.1 AI服务无法启动

**问题**: AI服务启动时卡住或报错

**可能原因与解决方案**:

1. **内存不足**
   ```bash
   # 检查系统内存
   free -h
   
   # 解决: 使用更轻量的模型或增加swap
   sudo fallocate -l 8G /swapfile
   sudo chmod 600 /swapfile
   sudo mkswap /swapfile
   sudo swapon /swapfile
   ```

2. **模型下载失败**
   ```bash
   # 手动下载模型
   cd ai-service/models
   
   # 使用镜像站下载
   export HF_ENDPOINT=https://hf-mirror.com
   huggingface-cli download sentence-transformers/paraphrase-multilingual-mpnet-base-v2
   ```

3. **CUDA版本不匹配**
   ```bash
   # 检查CUDA版本
   nvidia-smi
   
   # 安装对应版本的PyTorch
   pip install torch==2.0.1+cu118 -f https://download.pytorch.org/whl/torch_stable.html
   ```

## 总结

本部署文档涵盖了从本地开发环境到Docker生产环境的完整部署流程。

**推荐部署方案**:
- **开发环境**: 本地安装 + 依赖服务Docker化
- **测试/生产环境**: 完整Docker Compose部署 (使用 `docker-compose.mysql-qdrant.yml`)

**关键注意事项**:
1. AI服务首次启动需要下载模型，耗时较长 (10-30分钟)
2. 推荐使用Qdrant作为向量数据库 (ARM64兼容性好)
3. 生产环境务必修改默认密码
4. 定期备份MySQL和Qdrant数据
5. 监控系统资源使用，及时扩容

如有问题，请查看日志或联系技术支持。
