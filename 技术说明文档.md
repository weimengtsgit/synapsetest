### 3.1 智能测试用例生成

#### 3.1.1 技术方案：RAG + LLM

**核心思路**: 结合检索增强生成(Retrieval Augmented Generation)和大语言模型(Large Language Model)，实现高质量测试用例的智能生成。

**技术架构**:

```
需求文本输入
    ↓
文档解析与特征提取
    ↓
    ├─→ 向量数据库(Qdrant/Milvus) ←─ 语义检索
    │        ↓
    │   相似用例ID列表
    │        ↓
    └─→ MySQL数据库 ←─ 结构化查询
             ↓
        完整历史用例数据
             ↓
        企业测试标准加载
             ↓
        Prompt模板构建
             ↓
    LLM推理生成(Qwen/DeepSeek)
             ↓
        结构化测试用例输出
```

```python
def generate(
    self,
    requirement_text: str,
    module: str,
    num_cases: int = 5,
    include_edge_cases: bool = True
) -> Dict[str, Any]:
    """
    智能生成测试用例
    
    参数:
        requirement_text: 需求描述文本
        module: 所属模块
        num_cases: 生成用例数量
        include_edge_cases: 是否包含边界用例
    
    返回:
        生成的测试用例列表及元数据
    """
    
    # 步骤1: 解析需求文档，提取关键信息
    parsed_req = self._parse_requirement(requirement_text)
    
    # 步骤2: RAG检索相似历史用例
    similar_cases = self.rag_manager.search_similar_testcases(
        query_text=requirement_text,
        top_k=5,
        module_filter=module
    )
    
    # 步骤3: 加载企业测试标准
    standards = self._load_company_standards(module)
    
    # 步骤4: 构建Prompt
    prompt = self._build_prompt(
        requirement=parsed_req,
        similar_cases=similar_cases,
        standards=standards,
        num_cases=num_cases,
        include_edge_cases=include_edge_cases
    )
    
    # 步骤5: 调用LLM生成
    response = self.llm_client.generate(
        prompt=prompt,
        max_tokens=2000,
        temperature=0.7
    )
    
    # 步骤6: 解析并结构化输出
    testcases = self._parse_llm_response(response)
    
    return {
        "testcases": testcases,
        "metadata": {
            "requirement_id": parsed_req.get("id"),
            "module": module,
            "generation_time": datetime.now().isoformat(),
            "model_used": self.llm_client.model_name
        }
    }
```

**混合检索策略**:

```python
def search_similar_testcases(
    self,
    query_text: str,
    top_k: int = 5,
    module_filter: Optional[str] = None
) -> List[Dict[str, Any]]:
    """
    混合检索: 向量数据库 + MySQL
    
    步骤1: 向量数据库语义检索 (Qdrant/Milvus)
    步骤2: MySQL获取完整数据
    """
    
    # 1. 文本向量化
    query_embedding = self.embedding_model.encode(query_text)
    
    # 2. 向量数据库检索 (Qdrant)
    search_results = self.vector_db_client.search(
        collection_name="testcases",
        query_vector=query_embedding,
        limit=top_k,
        filter={
            "module": module_filter
        } if module_filter else None
    )
    
    # 3. 提取候选ID
    candidate_ids = [hit.id for hit in search_results]
    
    # 4. MySQL批量查询完整数据
    full_testcases = self.mysql_client.query(
        """
        SELECT id, title, description, precondition, 
               steps, expected_result, priority, module
        FROM testcases 
        WHERE id IN (%s)
        ORDER BY FIELD(id, %s)
        """ % (','.join(['%s'] * len(candidate_ids)), 
               ','.join(['%s'] * len(candidate_ids))),
        candidate_ids + candidate_ids
    )
    
    # 5. 合并相似度分数
    for i, testcase in enumerate(full_testcases):
        testcase['similarity_score'] = search_results[i].score
    
    return full_testcases
```

```python
def recommend(self, context: Dict[str, Any]) -> Dict[str, Any]:
    """
    智能推荐测试策略
    
    参数:
        context: 任务上下文信息
            - module: 模块名称
            - version_type: 版本类型 (major/minor/patch/hotfix)
            - change_scope: 变更范围 (small/medium/large)
            - risk_level: 风险等级 (low/medium/high/critical)
            - test_environment: 测试环境
    
    返回:
        推荐的测试策略
    """
    
    # 步骤1: 特征提取
    features = self._extract_features(context)
    
    # 步骤2: ML模型预测
    if self.model:
        ml_prediction = self._predict_with_model(features)
    else:
        ml_prediction = self._heuristic_prediction(features)
    
    # 步骤3: 规则引擎调整
    adjusted_strategy = self.rule_engine.apply_rules(
        prediction=ml_prediction,
        context=context
    )
    
    # 步骤4: 生成详细建议
    recommendations = self._generate_recommendations(
        strategy=adjusted_strategy,
        context=context
    )
    
    return {
        "strategy": adjusted_strategy,
        "recommendations": recommendations,
        "confidence": ml_prediction.get("confidence", 0.85)
    }
```

#### 3.2.3 特征工程

```python
def _extract_features(self, context: Dict[str, Any]) -> np.ndarray:
    """
    提取特征向量
    """
    features = []
    
    # 1. 模块复杂度特征
    module = context.get("module", "")
    module_complexity = self._get_module_complexity(module)
    features.append(module_complexity)
    
    # 2. 历史失败率
    historical_failure_rate = self._get_historical_failure_rate(module)
    features.append(historical_failure_rate)
    
    # 3. 变更规模编码
    change_scope_map = {"small": 0.3, "medium": 0.6, "large": 1.0}
    change_scope = change_scope_map.get(context.get("change_scope", "medium"), 0.6)
    features.append(change_scope)
    
    # 4. 风险等级编码
    risk_level_map = {"low": 0.25, "medium": 0.5, "high": 0.75, "critical": 1.0}
    risk_level = risk_level_map.get(context.get("risk_level", "medium"), 0.5)
    features.append(risk_level)
    
    # 5. 版本类型编码
    version_type_map = {"patch": 0.2, "minor": 0.5, "major": 0.8, "hotfix": 1.0}
    version_type = version_type_map.get(context.get("version_type", "minor"), 0.5)
    features.append(version_type)
    
    # 6. 测试覆盖率
    current_coverage = context.get("current_coverage", 0.7)
    features.append(current_coverage)
    
    # 7. 代码变更行数 (归一化)
    lines_changed = min(context.get("lines_changed", 100) / 1000.0, 1.0)
    features.append(lines_changed)
    
    # 8. 依赖模块数量 (归一化)
    dependencies = min(context.get("dependencies_count", 5) / 20.0, 1.0)
    features.append(dependencies)
    
    return np.array(features).reshape(1, -1)
```

#### 3.2.4 规则引擎

**文件位置**: `ai-service/models/recommendation/strategy_recommender.py`

**核心类**: `RuleEngine`

```python
class RuleEngine:
    """
    业务规则引擎
    """
    
    def apply_rules(
        self, 
        prediction: Dict[str, Any], 
        context: Dict[str, Any]
    ) -> Dict[str, Any]:
        """
        应用业务规则调整策略
        """
        strategy = prediction.copy()
        
        # 规则1: 关键模块必须全量回归
        if context.get("is_critical_module", False):
            strategy["test_scope"] = "full_regression"
            strategy["priority"] = "P0"
        
        # 规则2: 热修复必须快速验证
        if context.get("version_type") == "hotfix":
            strategy["test_scope"] = "targeted"
            strategy["execution_mode"] = "parallel"
            strategy["smoke_test_required"] = True
        
        # 规则3: 大规模变更增加测试范围
        if context.get("change_scope") == "large":
            if strategy["test_scope"] == "smoke":
                strategy["test_scope"] = "partial_regression"
            strategy["additional_checks"] = ["integration", "performance"]
        
        # 规则4: 高风险必须人工审核
        if context.get("risk_level") in ["high", "critical"]:
            strategy["manual_review_required"] = True
            strategy["approval_required"] = True
        
        # 规则5: 生产环境部署前必须UAT
        if context.get("target_environment") == "production":
            strategy["uat_required"] = True
            strategy["rollback_plan_required"] = True
        
        return strategy
```

### 3.3 智能去重

#### 3.3.1 技术方案：Sentence-BERT语义相似度

**核心思路**: 使用预训练的Sentence-BERT模型计算测试用例之间的语义相似度，通过聚类算法识别重复用例。

**技术架构**:

```
测试用例列表
    ↓
文本提取与预处理
    ├─→ 标题 (title)
    ├─→ 描述 (description)
    ├─→ 步骤 (steps)
    └─→ 预期结果 (expected_result)
    ↓
Sentence-BERT编码
    ↓
向量相似度矩阵计算 (Cosine Similarity)
    ↓
相似度阈值过滤 (threshold=0.85)
    ↓
聚类分组 (Connected Components)
    ↓
    ├─→ 保留用例 (每组选择最优)
    └─→ 重复用例 (标记为重复)
```

#### 3.3.2 实现细节

**文件位置**: `ai-service/models/optimization/deduplicator.py`

**核心类**: `SemanticDeduplicator`

**去重流程**:

```python
class SemanticDeduplicator:
    """
    基于语义相似度的智能去重
    """
    
    def __init__(
        self, 
        model_name: str = "paraphrase-multilingual-mpnet-base-v2",
        similarity_threshold: float = 0.85
    ):
        """
        初始化
        
        参数:
            model_name: Sentence-BERT模型名称
            similarity_threshold: 相似度阈值 (0.85表示85%相似即视为重复)
        """
        self.model = SentenceTransformer(model_name)
        self.similarity_threshold = similarity_threshold
    
    def deduplicate(
        self,
        testcases: List[Dict[str, Any]]
    ) -> Tuple[List[Dict[str, Any]], List[Dict[str, Any]]]:
        """
        执行智能去重
        
        返回:
            (保留的用例列表, 重复的用例列表)
        """
        
        # 步骤1: 文本提取
        texts = self._extract_texts(testcases)
        
        # 步骤2: 计算向量表示
        embeddings = self.model.encode(texts, convert_to_tensor=True)
        
        # 步骤3: 计算相似度矩阵
        similarity_matrix = util.cos_sim(embeddings, embeddings)
        
        # 步骤4: 识别重复组
        duplicate_groups = self._find_duplicate_groups(
            similarity_matrix, 
            self.similarity_threshold
        )
        
        # 步骤5: 选择保留用例
        kept_testcases, duplicate_testcases = self._select_best_from_groups(
            testcases, 
            duplicate_groups
        )
        
        return kept_testcases, duplicate_testcases
    
    def _extract_texts(self, testcases: List[Dict[str, Any]]) -> List[str]:
        """
        提取测试用例文本
        """
        texts = []
        for tc in testcases:
            # 拼接关键字段
            text_parts = [
                tc.get("title", ""),
                tc.get("description", ""),
                " ".join(tc.get("steps", [])),
                tc.get("expected_result", "")
            ]
            text = " ".join(filter(None, text_parts))
            texts.append(text)
        return texts
    
    def _find_duplicate_groups(
        self, 
        similarity_matrix: torch.Tensor,
        threshold: float
    ) -> List[List[int]]:
        """
        使用连通分量算法找出重复组
        """
        n = similarity_matrix.shape[0]
        
        # 构建邻接矩阵 (相似度超过阈值的视为连通)
        adj_matrix = (similarity_matrix > threshold).cpu().numpy()
        
        # 避免自连接
        np.fill_diagonal(adj_matrix, False)
        
        # 使用DFS找出所有连通分量
        visited = [False] * n
        groups = []
        
        def dfs(node, group):
            visited[node] = True
            group.append(node)
            for neighbor in range(n):
                if adj_matrix[node][neighbor] and not visited[neighbor]:
                    dfs(neighbor, group)
        
        for i in range(n):
            if not visited[i]:
                group = []
                dfs(i, group)
                if len(group) > 1:  # 只关注有重复的组
                    groups.append(group)
        
        return groups
    
    def _select_best_from_groups(
        self,
        testcases: List[Dict[str, Any]],
        duplicate_groups: List[List[int]]
    ) -> Tuple[List[Dict[str, Any]], List[Dict[str, Any]]]:
        """
        从每个重复组中选择最优用例
        
        选择策略:
        1. 优先级最高 (P0 > P1 > P2 > P3)
        2. 步骤最详细 (步骤数量最多)
        3. 创建时间最早 (更成熟)
        """
        kept_indices = set(range(len(testcases)))
        duplicate_testcases = []
        
        for group in duplicate_groups:
            # 评分排序
            scores = []
            for idx in group:
                tc = testcases[idx]
                score = self._calculate_quality_score(tc)
                scores.append((idx, score))
            
            # 按分数降序排序
            scores.sort(key=lambda x: x[1], reverse=True)
            
            # 保留最优，其余标记为重复
            best_idx = scores[0][0]
            for idx, _ in scores[1:]:
                kept_indices.discard(idx)
                tc_copy = testcases[idx].copy()
                tc_copy['duplicate_of'] = testcases[best_idx].get('id')
                duplicate_testcases.append(tc_copy)
        
        kept_testcases = [testcases[i] for i in sorted(kept_indices)]
        
        return kept_testcases, duplicate_testcases
    
    def _calculate_quality_score(self, testcase: Dict[str, Any]) -> float:
        """
        计算测试用例质量分数
        """
        score = 0.0
        
        # 优先级权重
        priority_weights = {"P0": 4.0, "P1": 3.0, "P2": 2.0, "P3": 1.0}
        score += priority_weights.get(testcase.get("priority", "P2"), 2.0) * 10
        
        # 步骤详细程度
        steps_count = len(testcase.get("steps", []))
        score += min(steps_count * 2, 20)
        
        # 描述完整性
        if testcase.get("description"):
            score += 10
        if testcase.get("precondition"):
            score += 5
        if testcase.get("expected_result"):
            score += 10
        
        # 执行历史 (有执行记录的更可靠)
        if testcase.get("execution_count", 0) > 0:
            score += 15
        
        return score
```

### 3.4 优先级智能排序

#### 3.4.1 技术方案：多因子加权评分

**核心思路**: 综合考虑业务价值、风险等级、执行成本、历史失败率、覆盖范围等多个维度，计算测试用例的优先级分数。

**评分模型**:

```
优先级分数 = 
    业务价值 × 0.30 +
    风险等级 × 0.25 +
    执行成本 × 0.20 +
    历史失败率 × 0.15 +
    覆盖影响 × 0.10
```

#### 3.4.2 实现细节

**文件位置**: `ai-service/models/optimization/prioritizer.py`

**核心类**: `TestCasePrioritizer`

**排序流程**:

```python
class TestCasePrioritizer:
    """
    测试用例智能优先级排序
    """
    
    def __init__(self, weights: Optional[Dict[str, float]] = None):
        """
        初始化
        
        参数:
            weights: 各因子权重配置
        """
        self.weights = weights or {
            "business_value": 0.30,
            "risk_level": 0.25,
            "execution_cost": 0.20,
            "failure_history": 0.15,
            "coverage_impact": 0.10
        }
    
    def prioritize(
        self, 
        testcases: List[Dict[str, Any]]
    ) -> List[Dict[str, Any]]:
        """
        执行智能排序
        
        返回:
            按优先级降序排列的测试用例列表
        """
        
        # 步骤1: 计算每个用例的优先级分数
        for tc in testcases:
            tc['priority_score'] = self._calculate_priority_score(tc)
        
        # 步骤2: 按分数降序排序
        sorted_testcases = sorted(
            testcases,
            key=lambda x: x['priority_score'],
            reverse=True
        )
        
        # 步骤3: 分配执行顺序
        for idx, tc in enumerate(sorted_testcases, start=1):
            tc['execution_order'] = idx
        
        return sorted_testcases
    
    def _calculate_priority_score(self, testcase: Dict[str, Any]) -> float:
        """
        计算优先级分数 (0-100)
        """
        
        # 因子1: 业务价值 (0-100)
        business_value = self._calculate_business_value(testcase)
        
        # 因子2: 风险等级 (0-100)
        risk_level = self._calculate_risk_level(testcase)
        
        # 因子3: 执行成本 (0-100, 成本越低分数越高)
        execution_cost = self._calculate_execution_cost(testcase)
        
        # 因子4: 历史失败率 (0-100)
        failure_history = self._calculate_failure_history(testcase)
        
        # 因子5: 覆盖影响 (0-100)
        coverage_impact = self._calculate_coverage_impact(testcase)
        
        # 加权求和
        score = (
            business_value * self.weights["business_value"] +
            risk_level * self.weights["risk_level"] +
            execution_cost * self.weights["execution_cost"] +
            failure_history * self.weights["failure_history"] +
            coverage_impact * self.weights["coverage_impact"]
        )
        
        return round(score, 2)
    
    def _calculate_business_value(self, testcase: Dict[str, Any]) -> float:
        """
        计算业务价值分数
        
        考虑因素:
        - 用例优先级标签 (P0/P1/P2/P3)
        - 关联功能的业务重要性
        - 用户使用频率
        """
        score = 0.0
        
        # 优先级标签映射
        priority_map = {
            "P0": 100,  # 核心功能
            "P1": 75,   # 重要功能
            "P2": 50,   # 一般功能
            "P3": 25    # 次要功能
        }
        priority = testcase.get("priority", "P2")
        score += priority_map.get(priority, 50)
        
        # 用户使用频率调整
        usage_frequency = testcase.get("usage_frequency", "medium")
        if usage_frequency == "high":
            score *= 1.2
        elif usage_frequency == "low":
            score *= 0.8
        
        return min(score, 100)
    
    def _calculate_risk_level(self, testcase: Dict[str, Any]) -> float:
        """
        计算风险等级分数
        
        考虑因素:
        - 代码变更影响范围
        - 模块稳定性
        - 历史问题严重程度
        """
        risk_map = {
            "critical": 100,
            "high": 80,
            "medium": 50,
            "low": 20
        }
        
        risk_level = testcase.get("risk_level", "medium")
        return risk_map.get(risk_level, 50)
    
    def _calculate_execution_cost(self, testcase: Dict[str, Any]) -> float:
        """
        计算执行成本分数 (成本越低，分数越高)
        
        考虑因素:
        - 执行时间
        - 环境依赖复杂度
        - 人工介入程度
        """
        # 执行时间 (秒)
        execution_time = testcase.get("estimated_duration", 60)
        
        # 时间越短，分数越高 (假设最长1小时)
        time_score = max(0, 100 - (execution_time / 36))
        
        # 自动化程度
        is_automated = testcase.get("is_automated", False)
        automation_bonus = 20 if is_automated else 0
        
        score = time_score + automation_bonus
        return min(score, 100)
    
    def _calculate_failure_history(self, testcase: Dict[str, Any]) -> float:
        """
        计算历史失败率分数
        
        考虑因素:
        - 历史执行次数
        - 历史失败次数
        - 最近失败时间
        """
        execution_count = testcase.get("execution_count", 0)
        failure_count = testcase.get("failure_count", 0)
        
        if execution_count == 0:
            return 30  # 未执行过的用例给予中等分数
        
        failure_rate = failure_count / execution_count
        
        # 失败率越高，分数越高 (这些用例更需要关注)
        score = failure_rate * 100
        
        # 最近失败加成
        last_failure = testcase.get("last_failure_time")
        if last_failure:
            days_since_failure = (datetime.now() - datetime.fromisoformat(last_failure)).days
            if days_since_failure <= 7:
                score *= 1.3  # 最近失败过，提高优先级
        
        return min(score, 100)
    
    def _calculate_coverage_impact(self, testcase: Dict[str, Any]) -> float:
        """
        计算覆盖影响分数
        
        考虑因素:
        - 代码覆盖范围
        - 功能覆盖广度
        - 依赖关系数量
        """
        score = 50  # 基础分
        
        # 覆盖的代码行数
        code_lines_covered = testcase.get("code_lines_covered", 0)
        score += min(code_lines_covered / 10, 30)
        
        # 依赖的模块数量
        dependencies_count = testcase.get("dependencies_count", 0)
        score += min(dependencies_count * 5, 20)
        
        return min(score, 100)
```

```python
def analyze_execution_quality(execution_data: Dict) -> Dict[str, Any]:
    """
    分析执行质量
    """
    total = execution_data["total_cases"]
    passed = execution_data["passed_cases"]
    failed = execution_data["failed_cases"]
    blocked = execution_data["blocked_cases"]
    
    # 计算各项指标
    pass_rate = (passed / total * 100) if total > 0 else 0
    fail_rate = (failed / total * 100) if total > 0 else 0
    block_rate = (blocked / total * 100) if total > 0 else 0
    
    # 计算执行效率
    avg_execution_time = execution_data.get("avg_execution_time", 0)
    
    # 评分逻辑
    score = pass_rate * 0.7  # 通过率占70%
    
    # 失败率惩罚
    if fail_rate > 20:
        score -= (fail_rate - 20) * 0.5
    
    # 阻塞率惩罚
    if block_rate > 10:
        score -= (block_rate - 10) * 1.0
    
    score = max(0, min(score, 100))
    
    return {
        "score": round(score, 2),
        "metrics": {
            "pass_rate": round(pass_rate, 2),
            "fail_rate": round(fail_rate, 2),
            "block_rate": round(block_rate, 2),
            "avg_execution_time": avg_execution_time
        },
        "status": "excellent" if score >= 90 else "good" if score >= 70 else "warning" if score >= 50 else "poor"
    }
```

```python
def sanitize_user_input(text: str) -> str:
    """
    防止Prompt注入攻击
    """
    # 移除特殊指令标记
    forbidden_patterns = [
        r"<\|.*?\|>",           # 特殊标记
        r"system:",             # 系统指令
        r"assistant:",          # 助手指令
        r"###\s*Instruction",   # 指令注入
    ]
    
    for pattern in forbidden_patterns:
        text = re.sub(pattern, "", text, flags=re.IGNORECASE)
    
    # 长度限制
    if len(text) > 10000:
        text = text[:10000]
    
    return text.strip()
```
